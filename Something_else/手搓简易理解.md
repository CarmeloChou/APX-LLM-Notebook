# Before_Model

```python
## Hook简例
class BeforeModel:
    def __init__(self):
        self.BEFORE_MODEL = self.create()

    def create(self):
        return []
    
    def get(self):
        for name in self.BEFORE_MODEL:
            print(name)

BEFORE_MODEL = BeforeModel()

def before_model(func):
    def wrapper(*args, **kargs):
        response = func(*args, **kargs)
        
        return response
    # 放外面而不是wrapper里可以避免重复注册
    # 放里面每当运行func函数时都会注册一次
    BEFORE_MODEL.BEFORE_MODEL.append(func)
    return wrapper

@before_model
def hook_test1():
    print("这是hook1")
    
@before_model
def hook_test2(input: str):
    print(f"模型接收到参数长度为{len(input)}")
    return input!= ''
```
